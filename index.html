오렌지보드 (MPU6050) 설정 가이드

1. 회로 연결 방법 (GY-521 & LED ↔ 오렌지보드)

VCC ➔ 5V
GND ➔ GND
SDA ➔ A4
SCL ➔ A5
녹색 LED ➔ D2
빨간 LED ➔ D3

2. 아두이노 스케치 코드 (Baud: 115200)

#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

Adafruit_MPU6050 mpu;

// LED 핀 정의 (D2: 약한 경보 / D3: 강한 경보)
const int PIN_GREEN_LED = 2;
const int PIN_RED_LED   = 3;

void setup() {
  Serial.begin(115200);
  while (!Serial) delay(10);

  pinMode(PIN_GREEN_LED, OUTPUT);
  pinMode(PIN_RED_LED, OUTPUT);
  digitalWrite(PIN_GREEN_LED, LOW);
  digitalWrite(PIN_RED_LED, LOW);

  if (!mpu.begin()) {
    while (1) delay(10);
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_2_G);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
}

void loop() {
  // 웹 앱에서 전달되는 LED 제어 명령어 처리 ('R': 빨간, 'G': 녹색, '0': 소등)
  if (Serial.available() > 0) {
    char cmd = Serial.read();
    if (cmd == 'R') {
      digitalWrite(PIN_RED_LED, HIGH);
      digitalWrite(PIN_GREEN_LED, LOW);
    } else if (cmd == 'G') {
      digitalWrite(PIN_RED_LED, LOW);
      digitalWrite(PIN_GREEN_LED, HIGH);
    } else if (cmd == '0') {
      digitalWrite(PIN_RED_LED, LOW);
      digitalWrite(PIN_GREEN_LED, LOW);
    }
  }

  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  // X, Y, Z 가속도(m/s²)를 쉼표로 구별하여 시리얼 전송
  Serial.print(a.acceleration.x);
  Serial.print(",");
  Serial.print(a.acceleration.y);
  Serial.print(",");
  Serial.println(a.acceleration.z);

  delay(20); // 50Hz
}
